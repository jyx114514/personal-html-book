代码示例 ​

python
import math

def threshold_cycle(initial, threshold, efficiency=1.0):
    return math.log(threshold / initial) / math.log1p(efficiency)

# 样式示例；本页没有执行 Python。
ct = threshold_cycle(initial=100, threshold=1_000_000)
